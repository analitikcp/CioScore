name: Build Android APK

on:
  workflow_dispatch:
  push:
    branches:
      - main

permissions:
  contents: read

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Java 17 kur
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - name: Gradle 9.5.1 kur
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '9.5.1'

      - name: Android projesini hazırla
        shell: bash
        run: |
          set -euo pipefail

          rm -rf /tmp/cio_project /tmp/cio_extract
          mkdir -p /tmp/cio_project /tmp/cio_extract

          echo "=== REPO ZIP'LERİ ==="
          find . -maxdepth 2 -type f -iname '*.zip' -not -path './.git/*' -print | sort

          # Öncelik: doğrudan proje ZIP'i.
          if [ -f "./CioScore_DIRECT_PROJECT.zip" ]; then
            SOURCE_ZIP="./CioScore_DIRECT_PROJECT.zip"
          elif [ -f "./CIO_SCORE_FINAL(2).zip" ]; then
            SOURCE_ZIP="./CIO_SCORE_FINAL(2).zip"
          else
            # Diğer ZIP'ler arasından ilk geçerli ZIP'i bul.
            SOURCE_ZIP=""
            while IFS= read -r -d '' ZIP; do
              rm -rf /tmp/cio_extract/*
              unzip -q -o "$ZIP" -d /tmp/cio_extract

              if find /tmp/cio_extract -type f \( -name 'settings.gradle.kts' -o -name 'settings.gradle' \) | grep -q . &&                  find /tmp/cio_extract -type f \( -path '*/app/build.gradle.kts' -o -path '*/app/build.gradle' \) | grep -q .; then
                SOURCE_ZIP="$ZIP"
                break
              fi
            done < <(find . -maxdepth 2 -type f -iname '*.zip' -not -path './.git/*' -print0)

            if [ -z "$SOURCE_ZIP" ]; then
              echo "HATA: Android proje ZIP'i bulunamadı."
              exit 1
            fi
          fi

          echo "=== KULLANILAN ZIP ==="
          echo "$SOURCE_ZIP"

          # TEK SEFERDE ve -o ile açılır; interaktif 'replace?' sorusu yok.
          rm -rf /tmp/cio_extract/*
          unzip -q -o "$SOURCE_ZIP" -d /tmp/cio_extract

          SETTINGS_FILE="$(find /tmp/cio_extract -type f \( -name 'settings.gradle.kts' -o -name 'settings.gradle' \) -print -quit)"
          if [ -z "$SETTINGS_FILE" ]; then
            echo "HATA: settings.gradle(.kts) bulunamadı."
            exit 1
          fi

          PROJECT_DIR="$(dirname "$SETTINGS_FILE")"

          if [ ! -f "$PROJECT_DIR/app/build.gradle.kts" ] && [ ! -f "$PROJECT_DIR/app/build.gradle" ]; then
            echo "HATA: $PROJECT_DIR/app/build.gradle(.kts) bulunamadı."
            exit 1
          fi

          cp -a "$PROJECT_DIR/." /tmp/cio_project/

          echo "PROJECT_DIR=/tmp/cio_project" >> "$GITHUB_ENV"

          echo ""
          echo "=== ANDROID PROJESİ ==="
          cd /tmp/cio_project
          pwd
          ls -la
          echo ""
          echo "=== KAYNAKLAR ==="
          find app/src/main -type f \( -name '*.kt' -o -name '*.java' \) -print | sort

      - name: APK derle
        shell: bash
        run: |
          set -euo pipefail
          cd "$PROJECT_DIR"

          echo "=== APK DERLEME ==="
          gradle --no-daemon --stacktrace :app:assembleDebug

      - name: APK kontrol
        shell: bash
        run: |
          set -euo pipefail

          echo "=== APK DOSYALARI ==="
          find "$PROJECT_DIR/app/build/outputs" -type f -name '*.apk' -print

          APK="$(find "$PROJECT_DIR/app/build/outputs" -type f -name '*.apk' -print -quit)"

          if [ -z "$APK" ]; then
            echo "HATA: APK oluşmadı!"
            exit 1
          fi

          echo "APK HAZIR:"
          ls -lh "$APK"

      - name: APK yükle
        uses: actions/upload-artifact@v4
        with:
          name: CIO-Score-APK
          path: ${{ env.PROJECT_DIR }}/app/build/outputs/apk/debug/*.apk
          if-no-files-found: error
